1"""

9 db = WaterDB(path)

11 for lat, lon in poly.vertices: 

12 ...

89class WaterDB: 

90 # Per-polygon vertex cap applied on load. OSM coastline polygons 

91 # routinely carry thousands of vertices at ~30 m spacing — far 

92 # finer than any 5" / 7" cockpit display can resolve at 30 NM 

93 # range. The CPU projection loop is per-vertex Python work, so 

94 # uncapped polygons make pyEfis unusable when the camera is near 

95 # a complex coastline. 64 vertices around any polygon is enough 

96 # to read the shape at typical SVS distances; tunable per-config 

97 # for users who want denser coastlines on larger displays. 

98 DEFAULT_MAX_VERTICES = 32

110 # scale queries against the 878k-polygon OSM dataset.

112 self._has_triangles = False 

113 self._has_rings = False

123 # wrong we drop back to "not ready" rather than crashing 

124 # downstream.

127 " kind, elev_ft, vertices " 

128 "FROM water_polygons LIMIT 0")

130 # it fall back to the plain bbox query. 

131 try:

135 log.info("WaterDB: using R-Tree spatial index") 

136 except sqlite3.OperationalError:

138 "WaterDB: no R-Tree found, falling back to bbox " 

139 "B-tree index (slower; rebuild with current " 

140 "build_water_db.py for R-Tree)") 

141 # Probe for the triangles column (added in the SVS-overlays- 

142 # to-GPU migration). Pre-tessellation DBs work too — the 

143 # renderer will fall back to a fan triangulation when the 

144 # column is missing. 

145 try: 

146 self._con.execute( 

147 "SELECT triangles FROM water_polygons LIMIT 0") 

148 self._has_triangles = True 

149 except sqlite3.OperationalError: 

150 log.info( 

151 "WaterDB: no triangles column found; GPU water " 

152 "renderer will fan-tessellate at draw time " 

153 "(slower; rebuild with current build_water_db.py " 

154 "for pre-tessellation)") 

155 # Probe for the rings column (the #44 island-hole fix). 

156 # Older DBs without it are all single-ring; islands inside 

157 # water polygons render as water until the pack is rebuilt. 

158 try: 

159 self._con.execute( 

160 "SELECT rings FROM water_polygons LIMIT 0") 

161 self._has_rings = True 

162 except sqlite3.OperationalError: 

163 log.info( 

164 "WaterDB: no rings column found; island holes in " 

165 "water polygons render as water (rebuild with " 

166 "current build_water_db.py — issue #44)") 

167 except Exception as e:

175 def polygons_in_range(self, ac_lat: float, ac_lon: float, 

176 range_nm: float, 

177 min_bbox_diag_deg: float | None = None, 

178 drop_ocean: bool = False): 

179 """Yield ``WaterPolygon`` for each polygon whose bounding box 

180 overlaps the aircraft's ``range_nm`` square. The yielded 

181 polygons may extend well beyond the range — the SVS clipper 

182 handles geometric clipping in screen space. 

183 

184 ``min_bbox_diag_deg`` rejects polygons whose bbox diagonal is 

185 below this threshold (in degrees) at the SQL level — useful 

186 for dropping sub-pixel ponds before paying the BLOB-decode and 

187 per-vertex projection cost. The check is done on the squared 

188 diagonal so the SQL stays multiplication-only. Ocean polygons 

189 are NEVER filtered by size — a coastline poly's bbox can be 

190 tiny (small bay, inlet) yet still cover the visible field — 

191 UNLESS ``drop_ocean`` is set. 

192 

193 ``drop_ocean`` excludes ``kind == 'ocean'`` polygons entirely and 

194 subjects everything else to the size floor. At wide zoom the terrain 

195 void-water already shows oceans, so dropping the (never-size-filtered, 

196 vertex-heavy) coastline here is the cheap way to keep only large inland 

197 lakes — e.g. the Great Lakes — for orientation. 

198 """ 

199 if not self.ready: 

200 return 

201 lat_cos = math.cos(math.radians(ac_lat)) 

202 range_deg_lat = range_nm / NM_PER_DEG_LAT 

203 range_deg_lon = range_deg_lat / max(lat_cos, 1e-6) 

204 lat_lo = ac_lat - range_deg_lat 

205 lat_hi = ac_lat + range_deg_lat 

206 lon_lo = ac_lon - range_deg_lon 

207 lon_hi = ac_lon + range_deg_lon 

208 min_d2 = (min_bbox_diag_deg * min_bbox_diag_deg 

209 if min_bbox_diag_deg else None) 

210 

211 if self._has_rtree: 

212 # R-Tree virtual-table overlap: sqlite walks the spatial 

213 # index and returns only the polygons whose bbox actually 

214 # overlaps the query bbox. Sub-millisecond vs ~50 ms for 

215 # the B-tree fallback at OSM dataset scale. 

216 tri_expr = ("p.triangles" if self._has_triangles 

217 else "NULL AS triangles") 

218 rings_expr = ("p.rings" if self._has_rings 

219 else "NULL AS rings") 

220 sql = ( 

221 "SELECT p.id, p.kind, p.elev_ft, p.vertices, " 

222 f" {tri_expr}, {rings_expr}, " 

223 " p.min_lat, p.max_lat, p.min_lon, p.max_lon " 

224 "FROM water_polygons p " 

225 "JOIN water_rtree r ON r.id = p.id " 

226 "WHERE r.min_lat <= ? AND r.max_lat >= ? " 

227 " AND r.min_lon <= ? AND r.max_lon >= ?") 

228 params = (lat_hi, lat_lo, lon_hi, lon_lo) 

229 if drop_ocean: 229 ↛ 230line 229 didn't jump to line 230 because the condition on line 229 was never true

230 sql += " AND p.kind != 'ocean'" 

231 if min_d2 is not None: 231 ↛ 232line 231 didn't jump to line 232 because the condition on line 231 was never true

232 size = ("((p.max_lat-p.min_lat)*(p.max_lat-p.min_lat) " 

233 "+ (p.max_lon-p.min_lon)*(p.max_lon-p.min_lon)) >= ?") 

234 sql += (f" AND ({size})" if drop_ocean 

235 else f" AND (p.kind = 'ocean' OR {size})") 

236 params = params + (min_d2,) 

237 cur = self._con.execute(sql, params) 

238 else: 

239 tri_expr = ("triangles" if self._has_triangles 

240 else "NULL AS triangles") 

241 rings_expr = ("rings" if self._has_rings 

242 else "NULL AS rings") 

243 sql = ( 

244 "SELECT id, kind, elev_ft, vertices, " 

245 f" {tri_expr}, {rings_expr}, " 

246 " min_lat, max_lat, min_lon, max_lon " 

247 "FROM water_polygons " 

248 "WHERE max_lat > ? AND min_lat < ? " 

249 " AND max_lon > ? AND min_lon < ?") 

250 params = (lat_lo, lat_hi, lon_lo, lon_hi) 

251 if drop_ocean: 251 ↛ 252line 251 didn't jump to line 252 because the condition on line 251 was never true

253 if min_d2 is not None:

255 "+ (max_lon-min_lon)*(max_lon-min_lon)) >= ?") 

256 sql += (f" AND ({size})" if drop_ocean 

257 else f" AND (kind = 'ocean' OR {size})") 

258 params = params + (min_d2,) 

259 cur = self._con.execute(sql, params) 

260 cap = self._max_vertices 

261 for r in cur: 

262 # The stored vertex count picks the triangle/ring index 

263 # dtype (uint16 <= 65535 vertices, uint32 beyond) — dense 

264 # multi-ring cells overflow uint16 and the builder mirrors 

265 # this exact rule, so the row is self-describing. 

266 n_verts = len(r["vertices"]) // 16 if r["vertices"] else 0 

267 rings = _decode_rings(r["rings"], n_verts) 

268 # Multi-ring rows are never re-decimated on load: stride 

269 # decimation across concatenated rings would corrupt the 

270 # ring topology AND the pre-tessellated triangle indices. 

271 # The build-time per-ring cap is the bound for these rows. 

272 yield WaterPolygon( 

273 id=r["id"], 

274 kind=(r["kind"] or "").strip().lower(), 

275 elev_ft=r["elev_ft"], 

276 vertices=_decode_vertices(r["vertices"], 

277 None if rings else cap), 

278 triangles=_decode_triangles(r["triangles"], n_verts), 

279 rings=rings)

282def _decode_vertices(blob: bytes, max_vertices: int | None = None) -> list: 

283 """Unpack a struct-packed <dd... bytes blob into [(lat, lon), ...]. 

284 

285 When ``max_vertices`` is set and the polygon has more vertices than 

286 that, stride-decimate uniformly down to the cap (always preserving 

287 the first and last points so the polygon's bounding extent is 

288 intact). This is the simplest possible cartographic simplification 

289 — Douglas-Peucker would preserve curvature better but isn't worth 

290 the per-load CPU when the screen can't resolve the difference at 

291 typical SVS distances. 

292 

293 Uses numpy.frombuffer for the bulk decode — at ~200 polygons / 

294 frame in the GPU water path each saved millisecond shows up in 

295 the visible frame rate, and the Python ``struct.unpack_from`` 

296 loop was visible in the profile.""" 

297 if not blob: 297 ↛ 298line 297 didn't jump to line 298 because the condition on line 297 was never true

298 return [] 

299 import numpy as _np 

300 pts = _np.frombuffer(blob, dtype="<f8").reshape(-1, 2) 

301 n = pts.shape[0] 

302 if max_vertices and n > max_vertices: 302 ↛ 304line 302 didn't jump to line 304 because the condition on line 302 was never true

303 # Even-stride decimation preserving indices 0 and n-1. 

304 idx = _np.round(_np.linspace(0, n - 1, max_vertices)).astype(int) 

305 pts = pts[idx] 

306 # Return as a list of plain Python tuples — the GPU collector 

307 # immediately re-wraps them with np.asarray, but the rest of the 

308 # codebase (and tests) treats vertices as a list-of-tuples. 

309 return [tuple(row) for row in pts.tolist()]

312def encode_vertices(vertices) -> bytes: 

313 """Pack [(lat, lon), ...] into a struct-packed <dd... bytes blob. 

314 Inverse of ``_decode_vertices``; used by the build tool.""" 

315 buf = bytearray(len(vertices) * 16) 

316 for i, (lat, lon) in enumerate(vertices): 

317 struct.pack_into("<dd", buf, i * 16, float(lat), float(lon)) 

318 return bytes(buf)
