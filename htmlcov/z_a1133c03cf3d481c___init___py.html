17"""Data Status instrument. 

18 

19Reads the navigation-data status JSON written by the ``pyefis-data`` updater 

20(Phase F of makerplane-data) and presents a boot-time Data Status screen: 

21per-pack currency with a subtle amber/white/healthy colour, a Continue 

22button (always available — the EFIS informs, it never restricts), and an 

23Update button that runs the updater over-the-air (USB sticks are picked up 

24automatically by the updater's udev hook; this screen reflects them on its 

25next refresh). 

26 

27The status file is read defensively: a missing or malformed file yields a 

28quiet "unavailable" state and never raises — the screen must never be the 

29thing that stops the aircraft from booting. 

30"""

132class _PackRow(QWidget): 

133 """A full-width touch target for one pack: a tap anywhere on the row toggles 

134 its checkbox. The checkbox (and labels) are made mouse-transparent so every 

135 tap lands on the row, not the tiny 26px indicator — a precise tap on the box 

136 is unreliable on the eglfs touchscreen. 

137 

138 Tap vs. scroll: toggle on *release*, and only when the pointer hasn't moved 

139 more than ``_TAP_SLOP`` px since press. Two-finger scrolling drags the 

140 synthesized pointer across the row (and scrolls the content under it), so it 

141 exceeds the slop and does NOT toggle; a clean tap stays put and selects. 

142 (The old code toggled on *press*, which fired before the gesture could be 

143 recognised as a scroll — so scrolling kept flipping whatever row a finger 

144 first landed on.)""" 

145 

146 _TAP_SLOP = 12 # px of pointer movement still counted as a tap 

147 

148 def __init__(self, checkbox, parent=None): 

149 super().__init__(parent) 

150 self._cb = checkbox 

151 self._press_pos = None 

152 

153 def _moved(self, event): 

154 if self._press_pos is None: 

155 return None 

156 d = event.position() - self._press_pos 

157 return abs(d.x()) + abs(d.y()) # manhattan distance 

158 

159 def mousePressEvent(self, event): 

160 self._press_pos = event.position() 

161 event.accept() 

162 

163 def mouseMoveEvent(self, event): 

164 # Once the pointer wanders past the slop it's a scroll/drag, not a tap; 

165 # drop the press so the upcoming release won't toggle. 

166 moved = self._moved(event) 

167 if moved is not None and moved > self._TAP_SLOP: 167 ↛ 169line 167 didn't jump to line 169 because the condition on line 167 was always true

168 self._press_pos = None 

169 event.accept() 

170 

171 def mouseReleaseEvent(self, event): 

172 moved = self._moved(event) 

173 if moved is not None and moved <= self._TAP_SLOP: 

174 self._cb.toggle() 

175 self._press_pos = None 

176 event.accept()
