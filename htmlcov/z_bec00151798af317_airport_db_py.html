1""" 

2Airport / runway database for the SVS renderer. 

3 

4Two backends, both exposing the same range-based query API: 

5 

6 db = AirportDB(...) 

7 for ap in db.airports_in_range(ac_lat, ac_lon, range_nm): 

8 ap.icao, ap.name, ap.ref_lat, ap.ref_lon, ap.elev_ft 

9 for rwy in ap.runways: 

10 rwy.thr1_lat, rwy.thr1_lon, rwy.thr1_elev_ft, ... 

11 

12* ``NASRAirportDB`` — reads ``airports.sqlite`` built from FAA NASR CSVs 

13 (see ``tools/build_airport_db.py``). Carries real runway widths, 

14 displaced-threshold offsets, marking type, TDZ elevation, approach 

15 lighting — everything needed to render Tier C surface markings. 

16 

17* ``CIFPAirportDB`` — wraps ``pyavtools.CIFPObjects`` (the same FAA CIFP 

18 data source used by Virtual VFR). Coarser detail: no width, no marking 

19 type, but works if a user only has the CIFP download. 

20 

21``make_airport_db(svs_config)`` picks the best available backend. 

22"""

185 "SELECT * FROM runway_ends WHERE site_no = ?", (site_no,)):

223def _designator_sort_key(end_id: str): 

224 """Sort key so '07' < '25' and 'L'/'C'/'R' suffixes stay grouped.""" 

225 s = (end_id or "").strip() 

226 digits = "".join(c for c in s if c.isdigit()) 

227 suffix = "".join(c for c in s if not c.isdigit()) 

228 try: 

229 n = int(digits) if digits else 99 

230 except ValueError: 

231 n = 99 

232 return (n, suffix)

273 def airports_in_range(self, ac_lat: float, ac_lon: float, 

274 range_nm: float): 

275 """Yield AirportRecord for each airport within *range_nm* of the 

276 aircraft. Skips airports with no matched runway pairs (which would 

277 produce nothing drawable anyway).""" 

278 if not self.ready: 

279 return 

280 lat_cos = math.cos(math.radians(ac_lat)) 

281 range_deg_lat = range_nm / NM_PER_DEG_LAT 

282 

283 # Load enough blocks to cover ±range_nm in both axes. CIFP blocks 

284 # are keyed by the degree-part of the latitude/longitude string 

285 # ("N34..."→34, "W119..."→-119), which Python's int() recovers via 

286 # truncation toward zero. (math.floor is wrong for negative 

287 # longitudes — floor(-119.84)=-120 but the index uses -119.) 

288 range_deg_lon = range_deg_lat / max(lat_cos, 1e-6) 

289 lat_min = int(ac_lat - range_deg_lat) 

290 lat_max = int(ac_lat + range_deg_lat) 

291 lon_min = int(ac_lon - range_deg_lon) 

292 lon_max = int(ac_lon + range_deg_lon) 

293 

294 # Collect all objects from the covered blocks, matching runways 

295 # across block boundaries as we go. 

296 airports: dict[str, AirportRecord] = {} 

297 unmatched_rwys: dict[str, list] = {} 

298 for block_lat in range(lat_min, lat_max + 1): 

299 for block_lon in range(lon_min, lon_max + 1): 

300 for obj in self._block(block_lat, block_lon): 

301 cls_name = obj.__class__.__name__ 

302 if cls_name == "Airport": 

303 if obj.id not in airports: 

304 airports[obj.id] = AirportRecord( 

305 icao=obj.id, name=obj.name, 

306 ref_lat=obj.lat, ref_lon=obj.lng, 

307 elev_ft=0.0) 

308 elif cls_name == "Runway": 

309 unmatched_rwys.setdefault(obj.airport_id, []).append(obj) 

310 

311 # Pair runways within each airport. CIFP runways come as one record 

312 # per threshold ("RW07" and "RW25" are the same physical strip). 

313 for icao, runways in unmatched_rwys.items(): 

314 consumed = set() 

315 for i, a in enumerate(runways): 

316 if i in consumed: 

317 continue 

318 # Find the matching opposing runway. 

319 partner_idx = None 

320 for j in range(i + 1, len(runways)): 

321 if j in consumed: 

322 continue 

323 if self._runways_match(a, runways[j]): 

324 partner_idx = j 

325 break 

326 if partner_idx is None: 

327 continue 

328 b = runways[partner_idx] 

329 consumed.add(i); consumed.add(partner_idx) 

330 

331 # Establish bearing thr1→thr2. CIFP carries each runway's 

332 # own bearing; use the lower-numbered end as thr1 so the 

333 # name "07" reads naturally. 

334 if (b.name or "").lstrip("RW") < (a.name or "").lstrip("RW"): 

335 a, b = b, a 

336 rec = RunwayRecord( 

337 airport_id=icao, name=(a.name or "").lstrip("RW"), 

338 thr1_lat=a.lat, thr1_lon=a.lng, 

339 thr1_elev_ft=float(a.elevation), 

340 thr2_lat=b.lat, thr2_lon=b.lng, 

341 thr2_elev_ft=float(b.elevation), 

342 length_ft=float(getattr(a, "length", 0) or 0), 

343 bearing_deg=float(getattr(a, "bearing", 0.0) or 0.0), 

344 ) 

345 ap = airports.get(icao) 

346 if ap is None: 

347 # CIFP sometimes has runways in a block whose airport 

348 # record is in a neighbouring block we haven't loaded. 

349 # Fabricate a minimal airport entry so the runway still 

350 # renders. 

351 ap = AirportRecord( 

352 icao=icao, name=icao, 

353 ref_lat=(a.lat + b.lat) * 0.5, 

354 ref_lon=(a.lng + b.lng) * 0.5, 

355 elev_ft=float(a.elevation)) 

356 airports[icao] = ap 

357 ap.runways.append(rec) 

358 

359 # Yield airports that ended up with at least one drawable runway and 

360 # whose reference point is actually within range_nm of the aircraft. 

361 range_nm_sq = range_nm * range_nm 

362 for ap in airports.values(): 

363 if not ap.runways: 

364 continue 

365 d_lat = (ap.ref_lat - ac_lat) * NM_PER_DEG_LAT 

366 d_lon = (ap.ref_lon - ac_lon) * NM_PER_DEG_LAT * lat_cos 

367 if d_lat * d_lat + d_lon * d_lon > range_nm_sq: 

368 continue 

369 # Use the first runway's threshold elevation as the field elev. 

370 ap.elev_ft = ap.runways[0].thr1_elev_ft 

371 yield ap

395 @staticmethod 

396 def _runways_match(a, b) -> bool: 

397 """Loose pair test that mirrors CIFPObjects.Runway.match() without 

398 raising on the empty-name records VVfr's loop catches with try/except. 

399 Two runways pair when they share an airport and their numbers 

400 differ by ~18 (180° apart). Letter suffixes (L/R/C) and the W/G 

401 helipad suffixes must agree.""" 

402 if a.airport_id != b.airport_id: 

403 return False 

404 na = (a.name or "").lstrip("RW") 

405 nb = (b.name or "").lstrip("RW") 

406 if not na or not nb: 

407 return False 

408 # Trailing letter suffix must match. 

409 def split(s): 

410 i = len(s) 

411 while i and not s[i-1].isdigit(): 

412 i -= 1 

413 return s[:i], s[i:] 

414 num_a, suf_a = split(na) 

415 num_b, suf_b = split(nb) 

416 if suf_a != suf_b: 

417 return False 

418 try: 

419 ia, ib = int(num_a), int(num_b) 

420 except ValueError: 

421 return False 

422 diff = abs(ia - ib) 

423 return diff == 18

433class _MultiAirportDB: 

434 """Merge several airport providers into one queryable DB. 

435 

436 Airport data comes from independent provider packs on their own update 

437 cadences — the US FAA NASR pack (primary, rich Tier-C data) plus optional 

438 supplemental packs (e.g. Canada from OurAirports). Each provider is a 

439 ``NASRAirportDB`` over its own sqlite. ``airports_in_range`` chains them and 

440 de-dups by ICAO so an airport that two providers both carry appears once 

441 (earlier providers — NASR first — win).""" 

442 

443 def __init__(self, dbs): 

444 self._dbs = [d for d in dbs if getattr(d, "ready", False)] 

445 

446 @property 

447 def ready(self) -> bool: 

448 return bool(self._dbs) 

449 

450 def airports_in_range(self, ac_lat, ac_lon, range_nm): 

451 seen = set() 

452 for db in self._dbs: 

453 for ap in db.airports_in_range(ac_lat, ac_lon, range_nm): 

454 key = (ap.icao or "").upper() or (round(ap.ref_lat, 4), 

455 round(ap.ref_lon, 4)) 

456 if key in seen: 

457 continue 

458 seen.add(key) 

459 yield ap

462def make_airport_db(config: dict): 

463 """Pick the best available airport DB backend for an SVS config. 

464 

465 Preference order: 

466 1. NASR primary (``nasr_db_path``) + any provider packs discovered under 

467 ``airport_provider_dir`` (``<dir>/*/current/airports.sqlite``), merged. 

468 2. CIFP if ``cifp_path`` and the index resolve. 

469 3. A no-op stub (``ready == False``). 

470 

471 Provider packs auto-discover: dropping a new airports-<x> pack into the 

472 provider dir makes its airports appear with no config change. 

473 """ 

474 paved = bool(config.get("paved_only", True)) 

475 dbs = [] 

476 nasr_path = config.get("nasr_db_path", "") 

477 if nasr_path: 

478 dbs.append(NASRAirportDB(nasr_path, paved_only=paved)) 

479 

480 provider_dir = config.get("airport_provider_dir", "") 

481 if provider_dir: 

482 import glob 

483 pat = str(Path(provider_dir) / "*" / "current" / "airports.sqlite") 

484 for p in sorted(glob.glob(pat)): 

485 dbs.append(NASRAirportDB(p, paved_only=paved)) 

486 

487 dbs = [d for d in dbs if getattr(d, "ready", False)] 

488 if len(dbs) == 1: 

489 return dbs[0] 

490 if dbs: 

491 return _MultiAirportDB(dbs) 

492 

493 cifp_path = config.get("cifp_path", "") 

494 index_path = config.get("cifp_index_path", "") 

495 if cifp_path and not index_path: 495 ↛ 496line 495 didn't jump to line 496 because the condition on line 495 was never true

496 index_path = str(Path(cifp_path).with_name("index.bin")) 

497 if cifp_path: 497 ↛ 498line 497 didn't jump to line 498 because the condition on line 497 was never true

498 db = CIFPAirportDB(cifp_path or None, index_path or None) 

499 if db.ready: 

500 return db
