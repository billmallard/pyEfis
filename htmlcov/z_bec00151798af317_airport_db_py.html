1""" 

2Airport / runway database for the SVS renderer. 

3 

4One backend, merged across provider packs, exposing a range-based query API: 

5 

6 db = make_airport_db(svs_config) 

7 for ap in db.airports_in_range(ac_lat, ac_lon, range_nm): 

8 ap.icao, ap.name, ap.ref_lat, ap.ref_lon, ap.elev_ft 

9 for rwy in ap.runways: 

10 rwy.thr1_lat, rwy.thr1_lon, rwy.thr1_elev_ft, ... 

11 

12* ``NASRAirportDB`` — reads ``airports.sqlite`` built from FAA NASR CSVs 

13 (see ``tools/build_airport_db.py``). Carries real runway widths, 

14 displaced-threshold offsets, marking type, TDZ elevation, approach 

15 lighting — everything needed to render Tier C surface markings. 

16 

17The ``CIFPAirportDB`` backend (``pyavtools.CIFPObjects``, the data path of 

18the legacy Virtual VFR overlay) was retired with that overlay (VVFR1, 

19AER-5417): the NASR pack is the only airport source. A config that still sets 

20``cifp_path`` gets a warning naming ``nasr_db_path`` and no airports. 

21 

22``make_airport_db(svs_config)`` merges whatever NASR-format packs resolve. 

23"""

48def _row_get(row, key, default=None): 

49 """Read an optional sqlite column, tolerating older airport packs built

105class NASRAirportDB: 

106 """SVS airport/runway data source backed by ``airports.sqlite`` — 

107 built once by ``tools/build_airport_db.py`` from the FAA NASR CSVs.""" 

108 

109 def __init__(self, sqlite_path: str | Path | None, paved_only=True): 

110 # Filter to paved runways (ASPH/CONC/PEM 

111 # substrings); airports left with no 

112 # runways drop entirely — flight-tested 

113 # declutter (DFW turf strips). 

114 self._paved_only = bool(paved_only) 

115 self._path = Path(sqlite_path) if sqlite_path else None 

116 self._con: sqlite3.Connection | None = None 

117 if self._path is None or not self._path.is_file(): 

118 log.info("NASRAirportDB: %s not found — falling back", self._path) 

119 return 

120 try: 

121 self._con = sqlite3.connect(str(self._path), 

122 check_same_thread=False) 

123 self._con.row_factory = sqlite3.Row 

124 except Exception as e: 

125 log.warning("NASRAirportDB: cannot open %s: %s", self._path, e) 

126 self._con = None 

127 

128 @property 

129 def ready(self) -> bool: 

130 return self._con is not None

132 def airports_in_range(self, ac_lat: float, ac_lon: float,

136 lat_cos = math.cos(math.radians(ac_lat)) 

137 range_deg_lat = range_nm / NM_PER_DEG_LAT 

138 range_deg_lon = range_deg_lat / max(lat_cos, 1e-6) 

139 lat_lo = ac_lat - range_deg_lat 

140 lat_hi = ac_lat + range_deg_lat 

141 lon_lo = ac_lon - range_deg_lon 

142 lon_hi = ac_lon + range_deg_lon 

143 

144 # First pass: airports within the bounding box. 

145 cur = self._con.execute( 

146 "SELECT site_no, icao, name, lat, lon, elev_ft " 

147 "FROM airports " 

148 "WHERE lat BETWEEN ? AND ? AND lon BETWEEN ? AND ?", 

149 (lat_lo, lat_hi, lon_lo, lon_hi)) 

150 airport_rows = cur.fetchall() 

151 if not airport_rows: 151 ↛ 152line 151 didn't jump to line 152 because the condition on line 151 was never true

152 return

154 range_nm_sq = range_nm * range_nm 

155 for ar in airport_rows:

157 d_lon = (ar["lon"] - ac_lon) * NM_PER_DEG_LAT * lat_cos

175 def _runways_for(self, site_no: str): 

176 # Fetch all runways and ends for this site, then pair each end pair 

177 # into a single RunwayRecord. 

178 rwys = self._con.execute( 

179 "SELECT rwy_id, length_ft, width_ft, surface FROM runways " 

180 "WHERE site_no = ?", (site_no,)).fetchall() 

181 if self._paved_only: 

182 rwys = [r for r in rwys 

183 if any(t in (r["surface"] or "").upper() 

184 for t in ("ASPH", "CONC", "PEM"))] 

185 ends_by_rwy: dict[str, list] = {} 

186 for e in self._con.execute( 

187 "SELECT * FROM runway_ends WHERE site_no = ?", (site_no,)): 

188 ends_by_rwy.setdefault(e["rwy_id"], []).append(e) 

189 

190 for r in rwys: 

191 ends = ends_by_rwy.get(r["rwy_id"], []) 

192 if len(ends) < 2: 192 ↛ 193line 192 didn't jump to line 193 because the condition on line 192 was never true

193 continue # need both ends to draw a quad 

194 # Sort by numeric designator so "07/25" yields end "07" first. 

195 ends.sort(key=lambda x: _designator_sort_key(x["end_id"])) 

196 a, b = ends[0], ends[1] 

197 if a["lat"] is None or b["lat"] is None: 197 ↛ 198line 197 didn't jump to line 198 because the condition on line 197 was never true

198 continue 

199 yield RunwayRecord( 

200 airport_id=site_no, name=r["rwy_id"], 

201 thr1_lat=a["lat"], thr1_lon=a["lon"], 

202 thr1_elev_ft=a["elev_ft"] or 0.0, 

203 thr2_lat=b["lat"], thr2_lon=b["lon"], 

204 thr2_elev_ft=b["elev_ft"] or 0.0, 

205 length_ft=r["length_ft"] or 0.0, 

206 bearing_deg=a["true_alignment_deg"] or 0.0, 

207 width_ft=r["width_ft"] or DEFAULT_RWY_WIDTH_FT, 

208 surface=(r["surface"] or "").strip(), 

209 thr1_designator=a["end_id"] or "", 

210 thr2_designator=b["end_id"] or "", 

211 thr1_marking=(a["marking_type"] or "").strip(), 

212 thr2_marking=(b["marking_type"] or "").strip(), 

213 thr1_displaced_ft=a["displaced_thr_len_ft"] or 0.0, 

214 thr2_displaced_ft=b["displaced_thr_len_ft"] or 0.0, 

215 thr1_tdz_elev_ft=a["tdz_elev_ft"] or 0.0, 

216 thr2_tdz_elev_ft=b["tdz_elev_ft"] or 0.0, 

217 thr1_apch_lgt=(a["apch_lgt_code"] or "").strip(), 

218 thr2_apch_lgt=(b["apch_lgt_code"] or "").strip(), 

219 thr1_vgsi_type=_row_get(a, "vgsi_code"), 

220 thr2_vgsi_type=_row_get(b, "vgsi_code"), 

221 thr1_visual_gpa=_row_get(a, "visual_gpa"), 

222 thr2_visual_gpa=_row_get(b, "visual_gpa"),

226def _designator_sort_key(end_id: str): 

227 """Sort key so '07' < '25' and 'L'/'C'/'R' suffixes stay grouped.""" 

228 s = (end_id or "").strip() 

229 digits = "".join(c for c in s if c.isdigit()) 

230 suffix = "".join(c for c in s if not c.isdigit()) 

231 try: 

232 n = int(digits) if digits else 99 

233 except ValueError: 

234 n = 99 

235 return (n, suffix)

241class _MultiAirportDB: 

242 """Merge several airport providers into one queryable DB. 

243 

244 Airport data comes from independent provider packs on their own update 

245 cadences — the US FAA NASR pack (primary, rich Tier-C data) plus optional 

246 supplemental packs (e.g. Canada from OurAirports). Each provider is a 

247 ``NASRAirportDB`` over its own sqlite. ``airports_in_range`` chains them and 

248 de-dups by ICAO so an airport that two providers both carry appears once 

249 (earlier providers — NASR first — win).""" 

250 

251 def __init__(self, dbs): 

252 self._dbs = [d for d in dbs if getattr(d, "ready", False)] 

253 

254 @property 

255 def ready(self) -> bool: 

256 return bool(self._dbs) 

257 

258 def airports_in_range(self, ac_lat, ac_lon, range_nm): 

259 seen = set() 

260 for db in self._dbs: 

261 for ap in db.airports_in_range(ac_lat, ac_lon, range_nm): 

262 key = (ap.icao or "").upper() or (round(ap.ref_lat, 4), 

263 round(ap.ref_lon, 4)) 

264 if key in seen: 

265 continue 

266 seen.add(key) 

267 yield ap

270def make_airport_db(config: dict): 

271 """Pick the best available airport DB backend for an SVS config. 

272 

273 Preference order: 

274 1. NASR primary (``nasr_db_path``) + any provider packs discovered under 

275 ``airport_provider_dir`` (``<dir>/*/current/airports.sqlite``), merged. 

276 2. A no-op stub (``ready == False``). 

277 

278 Provider packs auto-discover: dropping a new airports-<x> pack into the 

279 provider dir makes its airports appear with no config change. 

280 """ 

281 paved = bool(config.get("paved_only", True)) 

282 dbs = [] 

283 nasr_path = config.get("nasr_db_path", "") 

284 if nasr_path: 

285 dbs.append(NASRAirportDB(nasr_path, paved_only=paved)) 

286 

287 provider_dir = config.get("airport_provider_dir", "") 

288 if provider_dir: 

289 import glob 

290 pat = str(Path(provider_dir) / "*" / "current" / "airports.sqlite") 

291 for p in sorted(glob.glob(pat)): 

292 dbs.append(NASRAirportDB(p, paved_only=paved)) 

293 

294 dbs = [d for d in dbs if getattr(d, "ready", False)] 

295 if len(dbs) == 1: 

296 return dbs[0] 

297 if dbs: 

298 return _MultiAirportDB(dbs) 

299 

300 if config.get("cifp_path"): 

301 # VVFR1: the CIFP airport backend is gone. Say so rather than draw 

302 # no runways silently for a config written before the retirement. 

303 log.warning("AirportDB: cifp_path is no longer read (the CIFP " 

304 "airport backend was retired); set nasr_db_path to the " 

305 "airports.sqlite NASR pack") 

306 

307 # No data — return an empty stub so callers can rely on .ready. 

308 class _Empty: 

309 ready = False 

310 def airports_in_range(self, *a, **kw): 

311 return iter(()) 

312 return _Empty()
