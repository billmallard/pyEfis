2"""The `nav_status` instrument (FP7, billmallard/pyEfis#189): the GNX 375's 

3from-to-next field (Pilot's Guide 190-02488-01 Rev. B, 3-41/3-42) as a 

4one-line chip. 

5 

6 [LEG] KSBA > GVO > RZS 12.4 NM 0:06 TERM RNAV 2611 TURN R 

7 

8``WPFROM`` -> ``WPNAME`` (the TO waypoint, active colour) -> ``WPNEXT``, a 

9``FPLSTATE`` badge (LEG / DIRECT / SUSP, none while no plan is active), 

10optional ``WPDIS`` / ``WPETE`` and the flight phase (``FPLPHASE``, green; 

11yellow for ``LOI``). A missing ident is drawn as underscores. A tap fires the 

12``flightplan page fpl`` HMI action so the flight_plan instrument opens its FPL 

13page; ``tap_screen`` optionally switches to the screen that holds it first. 

14 

15The loaded procedure (PA9, billmallard/pyEfis#240): the approach type 

16(``FPLAPRTYPE``) and the AIRAC cycle the procedures came from (``FPLDBCYC``), 

17white -- or, once that cycle has expired, the cycle and ``EXPIRED`` in amber. 

18Expired data is annunciated here, on the approach, never refused (brief 

19guardrail 3). ``FPLTURN`` (fix-gateway, AER-4541) shows ``TURN L``/``TURN R`` 

20while the engine flies the coded turn that opens a DF leg. Both read the 

21engine's keys only: the direction is never derived from ``FPLfTURN`` and the 

22track, and a gateway without one of these keys shows nothing for it. 

23 

24Reads only engine outputs (fix-gateway ``flightplan`` plugin, brief section 

253.3); it owns no FIX keys and never writes. Construct-never-raises: a key the 

26gateway does not define is treated as permanently absent (the graceful- 

27missing-key convention in ``ai/__init__.py``), so the chip draws underscores 

28rather than failing the screen build. 

29 

30Quality: ``fail`` on a numeric field draws a red ``XXX`` in that field; 

31``old`` (the engine stopped publishing) greys the whole chip; ``bad`` greys a 

32field and ``WPETE`` bad (ground speed under 30 kt) is drawn as ``--:--``. 

33While ``FPLPHASE`` is ``VECTORS`` the engine publishes no distance or time 

34(it writes 0 with no flag), so both fields are blank rather than "0.0 NM". 

35"""

49# FPLSTATE (Appendix A): 0 NONE, 1 LEG, 2 DIRECT, 3 SUSP. 

50STATE_LABELS = {1: "LEG", 2: "DIRECT", 3: "SUSP"}

52NAV_KEYS = ("FPLSTATE", "FPLPHASE", "WPFROM", "WPNAME", "WPNEXT", "WPDIS", 

53 "WPETE") 

54# Newer than the chip: read when the gateway has them, never required (an 

55# older gateway leaves the chip available and the field blank). 

56OPTIONAL_KEYS = ("FPLAPRTYPE", "FPLDBCYC", "FPLTURN") 

57 

58MISSING_IDENT = "____" 

59FAIL_TEXT = "XXX" 

60BAD_ETE = "--:--" 

61SEPARATOR = ">" 

62TEMPLATE_IDENT = "WWWWW" 

63EXPIRED_TEXT = "EXPIRED" 

64TURN_TEXT = {"L": "TURN L", "R": "TURN R"}

67_FAIL = QColor("#ff0000") 

68_PHASE = QColor("#00ff00") 

69_PHASE_LOI = QColor("#ffff00") 

70# The data-status screen's expired colour (data_status SEVERITY_COLORS amber). 

71_EXPIRED = QColor(255, 176, 0) 

72_BADGE_COLORS = {"LEG": QColor("#ffffff"), "DIRECT": QColor("#ff00ff"),

76def format_distance(nm): 

77 return "%.1f NM" % nm if nm < 100.0 else "%.0f NM" % nm

80def format_ete(seconds): 

81 # H:MM, the flight_plan FPL page's ETE format (FlightPlan._fmt_ete). 

82 seconds = int(seconds) 

83 return "%d:%02d" % (seconds // 3600, (seconds % 3600) // 60)

86class NavStatus(QWidget): 

87 def __init__(self, parent=None, font_family="DejaVu Sans Condensed", 

88 font_percent=None): 

89 super(NavStatus, self).__init__(parent) 

90 self.font_family = font_family 

91 self.font_percent = font_percent 

92 # Panel options (InstrumentSpec Props in screenbuilder_factory). 

93 self.show_distance = True 

94 self.show_ete = True 

95 self.show_phase = True 

96 self.show_procedure = True 

97 self.show_turn = True 

98 self.hmi_group = "" 

99 self.tap_screen = "" 

100 self.active_color = "#ff00ff" 

101 self.text_color = "#ffffff"

103 self._items = {} 

104 for key in NAV_KEYS + OPTIONAL_KEYS: 

105 try: 

106 item = fix.db.get_item(key) 

107 except KeyError: 

108 logger.warning("nav_status: FIX key %s not defined by the " 

109 "gateway; the chip shows it as absent", key) 

110 continue 

111 self._items[key] = item 

112 item.valueChanged[item.dtype].connect(self._changed) 

113 item.oldChanged.connect(self._changed) 

114 item.badChanged.connect(self._changed) 

115 item.failChanged.connect(self._changed)

117 @property 

118 def available(self): 

119 return all(k in self._items for k in NAV_KEYS)

121 def _changed(self, *_): 

122 self.update()

124 # -- model ------------------------------------------------------------ 

125 def _read(self, key): 

126 """(value, old, bad, fail) for a key; an undefined key reads as 

127 (None, False, False, False).""" 

128 item = self._items.get(key) 

129 if item is None: 

130 return None, False, False, False 

131 return item.value, item.old, item.bad, item.fail

133 def fields(self): 

134 """The chip's content as a list of (text, QColor) runs plus the 

135 badge. Factored out of paintEvent so tests can assert what is drawn 

136 without scraping pixels.""" 

137 text = QColor(self.text_color) 

138 active = QColor(self.active_color) 

139 

140 state, state_old, _, _ = self._read("FPLSTATE") 

141 stale = bool(state_old) 

142 badge = STATE_LABELS.get(state) if state is not None else None 

143 

144 def ident(key): 

145 value, _, _, _ = self._read(key) 

146 value = (value or "").strip() if isinstance(value, str) else "" 

147 return value or MISSING_IDENT 

148 

149 # No active leg: all three are underscores, whatever stale idents the 

150 # keys may still carry. 

151 if not badge: 

152 idents = [MISSING_IDENT] * 3 

153 else: 

154 idents = [ident("WPFROM"), ident("WPNAME"), ident("WPNEXT")] 

155 

156 # Magenta means "the active TO waypoint"; underscores are not one. 

157 to_color = text if idents[1] == MISSING_IDENT else active 

158 runs = [(idents[0], text), (SEPARATOR, text), (idents[1], to_color), 

159 (SEPARATOR, text), (idents[2], text)] 

160 

161 phase, _, _, _ = self._read("FPLPHASE") 

162 phase = phase.strip() if isinstance(phase, str) else "" 

163 vectors = phase == "VECTORS" 

164 

165 data = [] 

166 if self.show_distance and badge and not vectors: 

167 dis, _, bad, fail = self._read("WPDIS") 

168 if fail: 

169 data.append((FAIL_TEXT, _FAIL)) 

170 elif dis is not None: 

171 data.append((format_distance(dis), _GREY if bad else text)) 

172 if self.show_ete and badge and not vectors: 

173 ete, _, bad, fail = self._read("WPETE") 

174 if fail: 

175 data.append((FAIL_TEXT, _FAIL)) 

176 elif bad: 

177 data.append((BAD_ETE, _GREY)) 

178 elif ete is not None: 

179 data.append((format_ete(ete), text)) 

180 if self.show_phase and phase: 

181 data.append((phase, _PHASE_LOI if phase == "LOI" else _PHASE)) 

182 if self.show_procedure and badge: 

183 proc = self.procedure_text() 

184 if proc: 

185 data.append(proc) 

186 if self.show_turn and badge: 

187 turn, _, _, _ = self._read("FPLTURN") 

188 turn = TURN_TEXT.get(turn.strip().upper() if isinstance(turn, str) else "") 

189 if turn: 

190 data.append((turn, text)) 

191 

192 if stale: 

193 runs = [(t, _GREY) for t, _ in runs] 

194 data = [(t, _GREY) for t, _ in data] 

195 return badge, runs, data, stale

197 def procedure_text(self, today=None): 

198 """(text, colour) for the loaded procedure, or None: ``RNAV 2611`` 

199 white; ``RNAV 2609 EXPIRED`` amber once the cycle has expired (UTC 

200 *today*, now by default); a departure or arrival with no approach 

201 shows the cycle alone. Nothing without a cycle on the bus.""" 

202 aprtype, _, _, _ = self._read("FPLAPRTYPE") 

203 cycle, _, _, _ = self._read("FPLDBCYC") 

204 aprtype = aprtype.strip() if isinstance(aprtype, str) else "" 

205 cycle = cycle.strip() if isinstance(cycle, str) else "" 

206 if not cycle: 

207 return None 

208 words = [w for w in (aprtype, cycle) if w] 

209 if airac.is_expired(cycle, today): 

210 return " ".join(words + [EXPIRED_TEXT]), _EXPIRED 

211 return " ".join(words), QColor(self.text_color)

214 def _font(self, px): 

215 font = QFont(self.font_family) 

216 font.setPixelSize(max(1, int(px))) 

217 font.setBold(True) 

218 return font

220 def _layout(self, fm, badge, runs, data): 

221 """Place every run left to right; returns ([(x, width, text, color, 

222 is_badge)], right edge). paintEvent lays out once at the nominal size 

223 to measure, then again at the fitted size to draw.""" 

224 space = fm.horizontalAdvance(" ") 

225 out = [] 

226 x = 0.0 

227 if badge: 

228 bw = fm.horizontalAdvance(badge) + space * 2 

229 out.append((x, bw, badge, None, True)) 

230 x += bw + space * 2 

231 for text, color in runs: 

232 adv = fm.horizontalAdvance(text) 

233 out.append((x, adv, text, color, False)) 

234 x += adv + space 

235 x += space * 2 

236 for text, color in data: 

237 adv = fm.horizontalAdvance(text) 

238 out.append((x, adv, text, color, False)) 

239 x += adv + space * 3 

240 return out, x - space * 3

242 def _template(self): 

243 """Worst-case content for the enabled fields. The chip is sized to fit 

244 this, not the live text, so the font does not jump as idents, badge 

245 and phase change in flight.""" 

246 runs = [(TEMPLATE_IDENT, None), (SEPARATOR, None)] * 2 + [(TEMPLATE_IDENT, None)] 

247 data = [] 

248 if self.show_distance: 248 ↛ 250line 248 didn't jump to line 250 because the condition on line 248 was always true

249 data.append(("888.8 NM", None)) 

250 if self.show_ete: 250 ↛ 252line 250 didn't jump to line 252 because the condition on line 250 was always true

251 data.append(("88:88", None)) 

252 if self.show_phase: 252 ↛ 257line 252 didn't jump to line 257 because the condition on line 252 was always true

253 data.append(("0.30 NM", None)) 

254 # The usual procedure, not the longest: an expired cycle or a long 

255 # type (LOCBC) is rare and shrinks the text further when it shows, 

256 # rather than shrinking every chip all the time. 

257 if self.show_procedure: 257 ↛ 259line 257 didn't jump to line 259 because the condition on line 257 was always true

258 data.append(("RNAV 8888", None)) 

259 if self.show_turn: 259 ↛ 261line 259 didn't jump to line 261 because the condition on line 259 was always true

260 data.append((TURN_TEXT["L"], None)) 

261 return "DIRECT", runs, data

263 def _fit_px(self, device, avail, h, badge, runs, data): 

264 """Pixel size for the chip: the nominal size (font_percent of 60% of 

265 the height), reduced until the worst-case template fits ``avail``, 

266 then further only if the live content is wider still (an ident 

267 longer than the template). Text width is close to linear in pixel 

268 size but not exactly (hinting), so the ratio step is followed by 1 px 

269 steps until it fits.""" 

270 scale = self.font_percent if self.font_percent else 1.0 

271 px = max(1.0, h * 0.6 * scale) 

272 for content in (self._template(), (badge, runs, data)): 

273 for _ in range(50): 273 ↛ 272line 273 didn't jump to line 272 because the loop on line 273 didn't complete

274 fm = QFontMetricsF(self._font(px), device) 

275 _, right = self._layout(fm, *content) 

276 if right <= avail or px <= 4.0: 

277 break 

278 px = max(4.0, min(px - 1.0, px * avail / right)) 

279 return px

281 def paintEvent(self, event): 

282 badge, runs, data, stale = self.fields() 

283 p = QPainter(self) 

284 p.setRenderHint(QPainter.RenderHint.Antialiasing) 

285 p.fillRect(self.rect(), QColor("#000000")) 

286 

287 w, h = float(self.width()), float(self.height()) 

288 margin = h * 0.15 

289 px = self._fit_px(p.device(), w - 2 * margin, h, badge, runs, data) 

290 font = self._font(px) 

291 p.setFont(font) 

292 fm = QFontMetricsF(font, p.device()) 

293 bh = fm.height() 

294 placed, _ = self._layout(fm, badge, runs, data) 

295 for x, adv, text, color, is_badge in placed: 

296 x += margin 

297 if is_badge: 

298 color = _GREY if stale else _BADGE_COLORS[text] 

299 rect = QRectF(x, (h - bh) / 2.0, adv, bh) 

300 p.setPen(QPen(color, max(1.0, h * 0.04))) 

301 p.setBrush(Qt.BrushStyle.NoBrush) 

302 p.drawRoundedRect(rect, bh * 0.2, bh * 0.2) 

303 p.drawText(rect, Qt.AlignmentFlag.AlignCenter, text) 

304 continue 

305 p.setPen(QPen(color)) 

306 p.drawText(QRectF(x, 0.0, adv + 1.0, h), 

307 Qt.AlignmentFlag.AlignLeft | Qt.AlignmentFlag.AlignVCenter, 

308 text) 

309 p.end()
