2# Moving map rivers layer (#92) -- OSM waterways as polylines. 

3# 

4# Rivers share the highways sqlite schema (same HighwayDB reader, built by 

5# tools/build_highway_db.py --preset rivers), so this is just RoadsLayer with 

6# its own class bands + colour + db option, per docs/map_layers_roadmap.md 

7# section 1.4. The worker-render-once + LOD-by-range machinery is inherited 

8# unchanged. This also gives the map the river-as-lines the SVS lacks (#39). 

9 

10from pyefis.instruments.map.layers import register_layer 

11from pyefis.instruments.map.layers.roads import RoadsLayer 

12 

13 

14@register_layer 

15class RiversLayer(RoadsLayer): 

16 id = "rivers" 

17 label = "Rivers" 

18 z = 9 # under roads (z=10), over terrain relief 

19 default_on = True 

20 

21 _DB_OPTION = "river_db_path" 

22 _COLOR_OPTION = "river_color" 

23 _DEFAULT_COLOR = "#4a6d8c" # muted water blue 

24 _EXPAND_LINKS = False # waterways have no _link ramps 

25 

26 # LOD bands (max range NM -> waterway classes). Major rivers persist to 

27 # wider range; canals/streams only close in. Keep in sync with 

28 # tools/build_highway_db.py PRESETS["rivers"]. 

29 _BAND_BASE = ( 

30 (8.0, ("river", "canal", "stream")), 

31 (25.0, ("river", "canal")), 

32 (60.0, ("river",)), 

33 ) 

34 _TIER = {"river": 0, "canal": 1, "stream": 2}
